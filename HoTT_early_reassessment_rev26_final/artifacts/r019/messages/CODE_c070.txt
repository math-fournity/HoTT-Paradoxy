0007 | class Term: pass

0021 | class App(Term):
0022 |     def __init__(self, f, arg): self.f = f; self.arg = arg

0025 | # --- 异化武器 1：单价公理 (将动作异化为空间) ---
0026 | class UA(Term):

0029 | class Transport(Term):
0030 |     def __init__(self, path, val): self.path = path; self.val = val

0033 | # --- 异化武器 2：命题截断与唯一选择 (将不停机异化为已完成) ---
0034 | class TruncProof(Term):

0037 | class UniqueChoice(Term):
0038 |     def __init__(self, trunc_proof): self.trunc_proof = trunc_proof

0043 | # 2. HoTT 静态逻辑审查 (Type Checker - 绕过 ASK)
0044 | # ==========================================
0045 | def hott_type_check(term):
0046 |     if isinstance(term, TrueVal) or isinstance(term, FalseVal):
0047 |         return Bool()
0048 |     elif isinstance(term, NotFunc):
0049 |         return "Bool -> Bool"
0050 |     elif isinstance(term, App):
0051 |         t_f = hott_type_check(term.f)
0052 |         t_arg = hott_type_check(term.arg)
0053 |         if t_f == "Bool -> Bool" and isinstance(t_arg, Bool): return Bool()
0054 |         raise Exception("Type Error")
0055 |         
0056 |     # 单价公理的类型赋予：直接发放合法通行证
0057 |     elif isinstance(term, Transport):
0058 |         return Bool() # 简写：transport 保证类型对齐
0059 |         
0060 |     # 唯一选择原则的类型赋予：从截断中直接提取出具体的自然数！
0061 |     elif isinstance(term, UniqueChoice):
0062 |         # 理论傲慢地宣布：这已经是一个合法的自然数了！
0063 |         return Nat()
0064 |     else:
0065 |         raise Exception(f"Unknown term: {term}")

0069 | # ==========================================

0072 |         f_eval = hott_evaluate(term.f)
0073 |         arg_eval = hott_evaluate(term.arg)
0074 |         if isinstance(f_eval, NotFunc):

0084 |     elif isinstance(term, UniqueChoice):
0085 |         return UniqueChoice(term.trunc_proof)
