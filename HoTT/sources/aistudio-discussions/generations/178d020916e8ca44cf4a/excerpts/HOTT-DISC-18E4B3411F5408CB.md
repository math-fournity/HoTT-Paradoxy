        return self.grid[ny][nx]

import collections

class RLEngine:
    def __init__(self, agent_controller):
        self.controller = agent_controller
        # Q-table: (state_tuple) -> [q_val_action_0, q_val_action_1, ...]
        self.q_table = collections.defaultdict(lambda: [0.0] * self.controller.output_size)
        self.learning_rate = 0.1
        self.discount_factor = 0.9
        self.epsilon = 0.1 # Exploration rate

    def _discretize_state(self, sensor_data):
        # 将连续的传感器数据离散化，以便用于Q-table
        return tuple(int(round(val * 10)) for val in sensor_data)

    def choose_action(self, state):
        if random.uniform(0, 1) < self.epsilon:
            return random.randint(0, self.controller.output_size - 1) # Explore
        else:
            q_values = self.q_table[state]
            return q_values.index(max(q_values)) # Exploit

    def learn(self, state, action, reward, next_state):
        old_value = self.q_table[state][action]
        future_optimal_value = max(self.q_table[next_state])
        
        # Q-learning formula
        new_value = old_value + self.learning_rate * (reward + self.discount_factor * future_optimal_value - old_value)
        self.q_table[state][action] = new_value

                if distance_sq > 0.1: # 避免除以零
                    distance = math.sqrt(distance_sq)
                    # 简化的引力 F = G*m_food / d^2, G=0.1
                    force_magnitude = 0.1 * food.mass / distance_sq
                    # a = F/m_agent
                    accel_x = force_magnitude * (dist_x / distance) / agent.mass
                    accel_y = force_magnitude * (dist_y / distance) / agent.mass
                    agent.vx += accel_x
                    agent.vy += accel_y

    def _handle_collisions(self):
        # 1.3.3 实现碰撞定律
        processed_pairs = set()
        all_objects = self._get_all_objects()

        for i in range(len(all_objects)):
            for j in range(i + 1, len(all_objects)):
                obj1 = all_objects[i]
                obj2 = all_objects[j]
                
                # Check if objects are in the same cell
                if int(obj1.x) == int(obj2.x) and int(obj1.y) == int(obj2.y):
                    pair = tuple(sorted((obj1.id, obj2.id)))
                    if pair not in processed_pairs:
                        # Agent-Food 碰撞
                        if isinstance(obj1, Agent) and isinstance(obj2, Food):
                            obj1.consume(obj2)
                            obj2.energy_value = -1 # Mark for removal
                        elif isinstance(obj2, Agent) and isinstance(obj1, Food):
                            obj2.consume(obj1)
                            obj1.energy_value = -1 # Mark for removal
                        
                        # Agent-Obstacle 碰撞
                        if isinstance(obj1, Agent) and isinstance(obj2, Obstacle):
                            obj1.vx *= -0.9 # Inelastic collision
                            obj1.vy *= -0.9
                        elif isinstance(obj2, Agent) and isinstance(obj1, Obstacle):
                            obj2.vx *= -0.9
                            obj2.vy *= -0.9
                            
                        processed_pairs.add(pair)
        
    def _cleanup_objects(self):
        # 清理被吃掉的食物
        for y in range(self.world.height):
            for x in range(self.world.width):
                self.world.grid[y][x] = [obj for obj in self.world.grid[y][x] if not (isinstance(obj, Food) and obj.energy_value == -1)]

    def run_step(self):
        self._apply_gravity()
        self._handle_collisions()
        self._cleanup_objects()
        self._update_positions()
        self.clock.tick()
