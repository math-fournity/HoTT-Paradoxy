                if abs(dist_y) > self.world.height / 2:

import collections

class RLEngine:
    def __init__(self, agent_controller):
        self.controller = agent_controller
        # Q-table: (state_tuple) -> [q_val_action_0, q_val_action_1, ...]
        self.q_table = collections.defaultdict(lambda: [0.0] * self.controller.output_size)
        self.learning_rate = 0.1
        self.discount_factor = 0.9
        self.epsilon = 0.1 # Exploration rate

    def _discretize_state(self, sensor_data):
        # 将连续的传感器数据离散化，以便用于Q-table
        return tuple(int(round(val * 10)) for val in sensor_data)

    def choose_action(self, state):
        if random.uniform(0, 1) < self.epsilon:
            return random.randint(0, self.controller.output_size - 1) # Explore
        else:
            q_values = self.q_table[state]
            return q_values.index(max(q_values)) # Exploit

    def learn(self, state, action, reward, next_state):
        old_value = self.q_table[state][action]
        future_optimal_value = max(self.q_table[next_state])
        
        # Q-learning formula
        new_value = old_value + self.learning_rate * (reward + self.discount_factor * future_optimal_value - old_value)
        self.q_table[state][action] = new_value

                dist_y = food.y - agent.y
                # 处理环绕空间的距离
                if abs(dist_x) > self.world.width / 2:
                    dist_x -= math.copysign(self.world.width, dist_x)
                if abs(dist_y) > self.world.height / 2:
